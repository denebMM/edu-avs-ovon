# docker-compose.yml - Versión corregida para que el orquestador vea los manifests

services:
  ollama:
    image: ollama/ollama:latest
    container_name: ollama_server
    ports:
      - "11434:11434"
    volumes:
      - ollama_models:/root/.ollama
    restart: unless-stopped
    networks:
      - ofp_network
    deploy:
      resources:
        limits:
          memory: 16G 
        reservations:
          devices:
            - driver: nvidia
              count: 1
              capabilities: [gpu]
    environment:
      - OLLAMA_NUM_GPU_LAYERS=-1
      - OLLAMA_KEEP_ALIVE=-1
      # --- AJUSTES PARA 16GB ---
      - OLLAMA_MAX_LOADED_MODELS=4   # Con 16GB puedes tener Llama 3.1, Gemma 2, Mistral y el clasificador cargados
      - OLLAMA_NUM_PARALLEL=4       # Permite procesar hasta 4 peticiones simultáneas en la GPU
      - OLLAMA_MAX_QUEUE=10         # Evita que las peticiones reboten si hay mucha carga
      
  orchestrator_ollama:
    build: ./orchestrator
    container_name: orchestrator_ollama
    ports:
      - "5000:5000"
    environment:
      - OLLAMA_HOST=http://ollama:11434
    depends_on:
      - ollama
    volumes:
      - ./schemas:/app/schemas
      - ./orchestrator/metrics:/app/metrics
    networks:
      - ofp_network

  orchestrator_manifest:
    build: ./orchestrator_manifest
    container_name: orchestrator_manifest
    ports:
      - "5004:5000"
    volumes:
      - ./schemas:/app/schemas
    depends_on:
      - assistant_physics
    networks:
      - ofp_network


  # ====================== ASISTENTE DE FÍSICA ======================
  physics-backend:
    build:
      context: ./assistants/physics-assistant/Physics-Chatbot-using-Advanced-RAG/backend
      dockerfile: Dockerfile
    container_name: physics-backend 
    command: python app.py                    
    environment:
      - OLLAMA_HOST=http://ollama:11434
      - PORT=5000                    # Puerto interno donde corre app.py
    volumes:
      - ./schemas:/app/schemas:ro
      - ./assistants/physics-assistant/Physics-Chatbot-using-Advanced-RAG/Data:/app/Data
      - ./assistants/physics-assistant/Physics-Chatbot-using-Advanced-RAG/Database:/app/Database
    networks:
      - ofp_network
    restart: unless-stopped

  assistant_physics:
    build:
      context: ./assistants/physics-assistant/Physics-Chatbot-using-Advanced-RAG/backend
      dockerfile: Dockerfile
    container_name: assistant_physics
    ports:
      - "5002:5002"
    command: python api_wrapper.py            
    environment:
      - PHYSICS_INTERNAL_URL=http://physics-backend:5000/api/generate   # ← CORREGIDO: puerto 8080
    depends_on:
      - physics-backend
    volumes:
      - ./schemas:/app/schemas:ro
    networks:
      - ofp_network
    restart: unless-stopped

  assistant_alice:

  assistant_history:

    # ====================== TRADUCTOR (Backend + Wrapper) ======================
  translator-backend:
    build:
      context: ./assistants/translator-assistant
      dockerfile: Dockerfile
    container_name: translator-backend
    command: python api_lingonaut.py
    ports:
      - "8081:8080"  # ← AÑADE ESTO (Mapeamos el 8080 interno al 8081 externo por si acaso)
    environment:
      - OLLAMA_HOST=http://ollama:11434
    volumes:
      - ./schemas:/app/schemas:ro
    networks:
      - ofp_network
    restart: unless-stopped

  assistant_translator:
    build:
      context: ./assistants/translator-assistant
      dockerfile: Dockerfile
    container_name: assistant_translator
    ports:
      - "5010:5010"
    # CAMBIAMOS ESTA LÍNEA: Usamos gunicorn para saltar la validación de host
    command: gunicorn --bind 0.0.0.0:5010 --timeout 180 api_wrapper:app
    environment:
      - LINGONAUT_BACKEND_URL=http://translator-backend:8080/chat
    depends_on:
      - translator-backend
    volumes:
      - ./schemas:/app/schemas:ro
    networks:
      - ofp_network
    restart: unless-stopped
